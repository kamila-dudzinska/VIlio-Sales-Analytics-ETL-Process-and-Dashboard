--liczba zamówień narastająco

liczba_zamówień_narastająco_rozpiska = var __max_data = MAX(Kalendarz[Date])               --data końcowa
                                var __min_data = DATE(YEAR(__max_data), 1, 1)    --data początkowa
                                var __zakres_dat = FILTER(                       --zakres dat
                                    All(Kalendarz[Date]), 
                                    Kalendarz[Date] >= __min_data &&            
                                    Kalendarz[Date] <= __max_data)
                                var __wynik = CALCULATE(                            --licz zamówieńia dla zakresu
                                                    [liczba_zamówień],
                                                    __zakres_dat)
                                Return __wynik  

--procent sprzedanych produktów powyżej średniej

%_sprzedanych_prod_pow_sredniej = 
                        var __licznik = [ile_prod_pow_sredniej]
                        var _liczba_prod = [liczba_sprzedanych_prod]
                        var __mianownik = CALCULATE(
                                            _liczba_prod, ALLSELECTED())
                        var __ulamek = DIVIDE(__licznik, __mianownik)
                        return __ułamek

-- procent dla danej kategrii w całości sprzedaży

%_w_całości_sprzedaży = 
                        var __licznik = SUM('OrderDetails (2)'[LineTotal])
                        var __mianownik = CALCULATE(
                                            SUM('OrderDetails (2)'[LineTotal]), ALLSELECTED())
                        var __ulamek = DIVIDE(__licznik, __mianownik)
                        return __ułamek

-- tabela kalendarza

Kalendarz = var __pierwsze_zam = MIN(Orders[Order Date])
            var __ostatnie_zam = MAX(Orders[Delivery Date])
            var __first_date = DATE(YEAR(__pierwsze_zam), 1, 1)
            var __last_date = DATE(YEAR(__ostatnie_zam), 12, 31)
            var __cal_V1 = CALENDAR(__first_date, __last_date)
            var __cal_final = ADDCOLUMNS(__cal_V1, 
                                        "rok", YEAR([Date]),
                                        "miesiąć", MONTH([Date]),
                                        "nazwa mieisaca", FORMAT([Date], "MMMM"),
                                        "dzień", DAY([Date]),
                                        "nazwa_dnia", FORMAT([Date], "DDDD"))
            return __cal_final
